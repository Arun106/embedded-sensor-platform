cmake_minimum_required(VERSION 3.22)
project(EmbeddedSensorGateway VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(APP_VERSION "${PROJECT_VERSION}" CACHE STRING "Release version")
add_executable(sensor-gateway app/main.cpp)
target_include_directories(sensor-gateway PRIVATE app)
target_compile_definitions(sensor-gateway PRIVATE APP_VERSION="${APP_VERSION}")
target_compile_options(sensor-gateway PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(sensor-unit-tests app/unit_tests.cpp)
  target_include_directories(sensor-unit-tests PRIVATE app)
  target_compile_options(sensor-unit-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME temperature_input__critical__unit_boundaries COMMAND sensor-unit-tests)
endif()
install(TARGETS sensor-gateway RUNTIME DESTINATION bin)
